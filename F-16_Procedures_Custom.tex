\begin{checklist}{Before Starting Engine}
	\item{COCKPIT INTERIOR CHECK}{Perform}
	\condition{EXT LIGHTING Panel} {
		\item{MASTER COVERT}{Norm}
		\item{ANTI COLLISION}{Off}
		\item{POSITION}{Steady}
		\item{WING/TAIL}{Brt}
		\item{FUSELAGE}{Off}
		\item{FORMATION}{100\%}
	}
	\item{C\&I}{BACKUP}
	\item{MASTER FUEL}{Check MASTER}
	\item{ENG FEED}{NORM}
	\condition{Activate Battery} {
		\item{MAIN PWR}{Batt}
		\item{ELEC SYS, FLCS RLY, MAIN \& STBY GEN Lights}{Check On}
		\item{FLCS PWR TEST}{Hold at Test}
		\item{FLCS PMG, TO FLCS, FLCS PWR 4 Lights}{Check On}
		\item{FLCS RLY}{Check Off}
		\item{FLCS PWR TEST}{Release}
	}
	\condition{Audio Setup}{
		\item{COMM1 \& 2}{Set as desired}
		\item{COMM1 \& 2}{Verify SQL}
		\item{MSL \& THREAT Volume}{Set > 50\%}
		\item{ILS}{Set on, as desired}
		\item{INTERCOM}{Set 100\%}
	}
	\condition{Backup UHF Setup}{
		\item{Left Function Knob}{BOTH}
		\item{Right Function Knob}{As Desired}
	}
	\item{Interior Lighting Panel}{Set as desired}
	\item{AIR SOURCE}{NORM}
	\item{ANTI ICE}{ON}
	\condition{Activate Main Power}{
		\item{MAIN PWR}{MAIN PWR}
		\item{ELEC SYS, FLCS RLY, MAIN \& STBY GEN Lights}{Check On}
		\item{SEC Caution, ENGINE \& HYD/OIL PRESS Warning Lights}{Check On}
	}
	\item{EPU GEN \& PMG Lights}{Check Off}
	\item{Canopy}{Close \& Lock}
\end{checklist}

\begin{checklist}{Engine Start}
	\item{Throttle}{Check Cut-Off}
	\item{JFS Switch}{Select Start 2}
	\item{JFS RUN Light}{Verify Illuminated within 30s}
	\item{SEC Caution Light}{Verify Extinguish at 20\% RPM}
	\item{Throttle}{Set Idle at 25\%RPM}
	\item{HYD/OIL PRESS Light}{Verify Extinguish at 30 - 35\% RPM}
	\item{FTIT}{Monitor, Must be Below 750\degree\ C}
	\note{FTIT should roughly stay 10x the RPM until it falls again.}
	\item{FLCS PMG Light}{Verify Extinguish at 40 - 45\% RPM}
	\item{JFS}{Verify Off at ca. 55\% RPM}
	\item{ENGINE, STBY GEN Light}{Verify Extinguish at ca. 60\% RPM}
	\item{MAIN GEN Light}{Verify Extinguish ca. 10s later}
	\condition{Idle Checks PW 229 / Block 52} {
		\item{FUEL FLOW}{Check 500 - 2000 pph}
		\item{NOZ POS}{Check > 80\%}
		\item{FTIT}{Check 625\degree\ C or less, abort if over 800\degree\ C}
		\item{HYD PRESS A \& B}{Check needles are vertical}
	}
	\condition{Idle Checks GE 129 / Block 50} {
		\item{FUEL FLOW}{Check 700 - 1700 pph}
		\item{NOZ POS}{Check > 94\%}
		\item{FTIT}{Check 650\degree\ C or less, abort if over 800\degree\ C}
		\item{HYD PRESS A \& B}{Check needles are vertical}
	}
	\condition{Idle Checks PW 220 / Block 42} {
		\item{FUEL FLOW}{Check 500 - 1500 pph}
		\item{NOZ POS}{Check 70 - 95\%}
		\item{FTIT}{Check 575\degree\ C or less, abort if over 800\degree\ C}
		\item{HYD PRESS A \& B}{Check needles are vertical}
	}
	\condition{Idle Checks GE 100 / Block 40} {
		\item{FUEL FLOW}{Check 700 - 1700 pph}
		\item{NOZ POS}{Check > 94\%}
		\item{FTIT}{Check 650\degree\ C or less, abort if over 800\degree\ C}
		\item{HYD PRESS A \& B}{Check needles are vertical}
	}
	\item{Gear Lights}{Check 3 Green}
\end{checklist}

\begin{checklist}{After Engine Start}
	\condition{FIRE \& OHEAT DETECT Test} {
		\item{FIRE \& OHEAT DETECT}{Press and Hold}
		\item{ENG FIRE, OVERHEAT, MASTER CAUTION Lights}{Check On}
		\item{FIRE \& OHEAT DETECT}{Release}
		\item{ENG FIRE, OVERHEAT Lights}{Check Off}
	}
	\condition{MAL \& IND LTS Test} {
		\item{MAL \& IND LTS}{Press and Hold}
		\item{All Lights}{Check On}
		\item{Voice Message System}{Verify Operation}
		\item{MAL \& IND LTS}{Release}
	}
	\condition{PROBE HEAT Test} {
		\item{PROBE HEAT Switch}{PROBE HEAT}
		\item{PROBE HEAT Caution Light}{Verify Off}
		\item{PROBE HEAT Switch}{Test}
		\item{PROBE HEAT Caution Light}{Verify Flashing}
		\item{PROBE HEAT Switch}{Off}
	}
	\condition{EPU Check} {
		\item{EPU Safety Pin}{Remove}
		\item{EPU}{Cycle Off, then Norm}
		\item{EPU GEN, EPU PMG Lights}{Check Off}
		\item{Throttle}{Set 80\% RPM}
		\item{EPU/GEN Switch}{Hold}
		\item{EPU AIR Light}{Check On}
		\item{EPU GEN, EPU PMG Lights}{Check Off}
		\item{EPU RUN and FLCS PWR 4 Lights}{On after 5s}
		\item{EPU/GEN Switch}{Release}
		\item{Throttle}{Set Idle}
		\note{If the green run light doesn't appear, reinitialise the test with higher throttle.}
	}
\end{checklist}

\begin{checklist}{Avionics \& Checks}
	\condition{FLCS BIT}{
		\item{Flight Controls}{Cycle for 20s}
		\item{FLCS Panel}{Check All Switches Down}
		\item{FLCS BIT}{Start}
		\note{If the BIT fails, repeat the three steps.}
	}
	\note{Continue while BIT is running.}
	\condition{Air Refueling Check - If refueling might be necessary}{
		\item{AIR REFUEL}{Open}
		\item{Right Indexer}{Check Blue RDY Light Illuminated}
		\item{Stick DISC Button}{Press}
		\item{Right Indexer}{Check red DISC Light Illuminated for 3s}
		\item{AIR REFUEL}{Close}
		\item{Right Indexer}{Check Extinguished}
	}
	\item{Audio Panels}{Check Again}
\end{checklist}
